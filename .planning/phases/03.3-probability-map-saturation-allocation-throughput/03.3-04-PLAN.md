  compute_per_transition_metrics(
    filtered_df,         # data.frame with at least: id_trans, From*, To*, Rate
    prob_map_dir,        # path to directory containing %03d_id_trans_%d.tif files
    anterior_path,       # path to anterior LULC raster
    posterior_path,      # path to posterior LULC raster
    threshold,           # numeric (e.g. 0.90)
    exempt               # list of list(from_lulc=int, to_lulc=int, reason=text)
  ) returns data.frame with EXACTLY these 16 columns (in this order):
    id_trans, from_val, to_val, rate,
    demanded_cells, placed_cells, placed_frac,
    coverage, p50, p90, p95, p99, pmax,
    demand_vs_capacity, exempt, floor_met

  - demanded_cells = round(rate * sum(anterior_vals == from_val, na.rm = TRUE))
  - placed_cells   = sum(anterior_vals == from_val & posterior_vals == to_val, na.rm = TRUE)
  - placed_frac    = placed_cells / demanded_cells (NA if demanded_cells == 0)
  - coverage       = sum(prob_vals > 0, na.rm = TRUE) / sum(!is.na(prob_vals))
  - p50,p90,p95,p99 = stats::quantile(prob_vals, c(0.5, 0.9, 0.95, 0.99), na.rm = TRUE)
  - pmax           = max(prob_vals, na.rm = TRUE)
  - demand_vs_capacity = demanded_cells / sum(prob_vals >= p50, na.rm = TRUE)
  - exempt         = TRUE if (from_val, to_val) matches any exempt list entry
  - floor_met      = !is.na(placed_frac) && placed_frac >= threshold
